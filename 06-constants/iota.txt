[iota로 열거값 사용]

const (
    Red int = iota
    Blue int = iota
    Grenn int = iota
)

const (
    C1 uint = iota + 1 // 1 = 0 + 1
    C2                 // 2 = 1 + 1
    C3                 // 3 = 2 + 1
)

[비트플래그로 응용]

const (
    BitFlag1 uint = 1 << iota // 1 = 1 << 0, 2진수로 0000 0001
    BitFlag2                  // 2 = 1 << 1, 2진수로 0000 0010
    BitFlag3                  // 4 = 1 << 2, 2진수로 0000 0100
    BitFlag4                  // 8 = 1 << 3, 2진수로 0000 1000
)
